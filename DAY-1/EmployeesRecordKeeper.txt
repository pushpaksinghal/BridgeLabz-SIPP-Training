CLASS Employee
    ATTRIBUTE name
    ATTRIBUTE department

    METHOD Constructor(name, department)
        SET this.name = name
        SET this.department = department

    METHOD print()
        PRINT "Name: " + name + ", Department: " + department

LIST employees
ADD NEW Employee("John", "HR") TO employees
ADD NEW Employee("Jane", "IT") TO employees
ADD NEW Employee("Tom", "IT") TO employees

LIST itEmployees

FOR EACH employee IN employees
    IF employee.department == "IT" THEN
        ADD employee TO itEmployees

FOR EACH employee IN itEmployees
    CALL employee.print()
